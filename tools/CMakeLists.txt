cmake_minimum_required(VERSION 3.20)
project(soldat_tools)
include(ExternalProject)

set(THIRD_PARTY_SRC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/src")
set(BUILD_ALWAYS_TOOLS ON)

set(DXC_ROOT ${CMAKE_CURRENT_BINARY_DIR}/dxc)
include(${THIRD_PARTY_SRC_DIR}/SDL_shadercross/build-scripts/download-prebuilt-DirectXShaderCompiler.cmake)

function(create_import_cmake_file)
  # Parse the arguments
  set(options "")
  set(oneValueArgs "")
  set(multiValueArgs PROJECTS)
  cmake_parse_arguments(ARG "${options}" "${oneValueArgs}" "${multiValueArgs}" ${ARGN})

  # Validate input
  if(NOT ARG_PROJECTS)
    message(FATAL_ERROR "PROJECTS parameter is required")
  endif()

  set(THIRD_PARTY_CMAKE ${CMAKE_BINARY_DIR}/install/tools.cmake)
  file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/install)
  file(WRITE ${THIRD_PARTY_CMAKE} "# Autogenerated file, do not edit\n")

  # Process each project in the list
  foreach(proj ${ARG_PROJECTS})
    ExternalProject_Get_Property(${proj} INSTALL_DIR)
    file(APPEND ${THIRD_PARTY_CMAKE}
      "list(APPEND CMAKE_PREFIX_PATH ${INSTALL_DIR})\n"
    )
  endforeach()
  file(APPEND ${THIRD_PARTY_CMAKE}
    "set(DirectXShaderCompiler_ROOT ${DXC_ROOT})"
  )
endfunction()

find_program(CCACHE_FOUND ccache)
if(CCACHE_FOUND)
   set_property(GLOBAL PROPERTY CXX_COMPILER_LAUNCHER ccache)
   set_property(GLOBAL PROPERTY C_COMPILER_LAUNCHER ccache)
endif()

if (CCACHE_FOUND)
  set(COMPILER_LAUNCHER "-DCMAKE_CXX_COMPILER_LAUNCHER=ccache;-DCMAKE_C_COMPILER_LAUNCHER=ccache")
else()
  set(COMPILER_LAUNCHER "")
endif()

set(COMMON_FLAGS
  ${COMPILER_LAUNCHER}
  -DCMAKE_INSTALL_PREFIX=<INSTALL_DIR>
  -DCMAKE_BUILD_TYPE=Release
)


### Tools

ExternalProject_Add(
  SDL3
  SOURCE_DIR "${THIRD_PARTY_SRC_DIR}/SDL"
  CMAKE_ARGS
    -DSDL_LIBURING=OFF
    -DSDL_INSTALL=ON
    -DSDL_UNINSTALL=OFF
    -DSDL_CCACHE=ON
    -DSDL_TEST_LIBRARY=OFF
    ${COMMON_FLAGS}
  UPDATE_COMMAND ""   # Not doing updates
  PATCH_COMMAND ""    # No patches
  USES_TERMINAL_CONFIGURE ON
  USES_TERMINAL_BUILD ON
  USES_TERMINAL_INSTALL ON
  BUILD_ALWAYS ${BUILD_ALWAYS_TOOLS}
)

ExternalProject_Get_Property(SDL3 INSTALL_DIR)
set(sdl3_install_dir ${INSTALL_DIR})


ExternalProject_Add(
  SPIRV-Cross
  SOURCE_DIR "${THIRD_PARTY_SRC_DIR}/SDL_shadercross/external/SPIRV-Cross"
  CMAKE_ARGS
    -DSPIRV_CROSS_SHARED=ON
    -DSPIRV_CROSS_STATIC=ON
    ${COMMON_FLAGS}
  UPDATE_COMMAND ""   # Not doing updates
  PATCH_COMMAND ""    # No patches
  USES_TERMINAL_CONFIGURE ON
  USES_TERMINAL_BUILD ON
  USES_TERMINAL_INSTALL ON
  BUILD_ALWAYS ${BUILD_ALWAYS_TOOLS}    
)

ExternalProject_Get_Property(SPIRV-Cross INSTALL_DIR)
set(spirv_cross_install_dir ${INSTALL_DIR})

ExternalProject_Add(
  SDL_shadercross DEPENDS SDL3 SPIRV-Cross
  SOURCE_DIR "${THIRD_PARTY_SRC_DIR}/SDL_shadercross"
  LIST_SEPARATOR |
  CMAKE_ARGS
    "-DCMAKE_PREFIX_PATH=${sdl3_install_dir}|${spirv_cross_install_dir}"
    -DSDLSHADERCROSS_DXC=ON
    -DSDLSHADERCROSS_VENDORED=OFF
    -DHLSL_COPY_GENERATED_SOURCES=ON
    -DSDLSHADERCROSS_INSTALL=ON
    -DDXC_ROOT=<INSTALL_DIR>
    -DDirectXShaderCompiler_ROOT=${DXC_ROOT}
    ${COMMON_FLAGS}
  UPDATE_COMMAND ""   # Not doing updates
  PATCH_COMMAND ""    # No patches
  USES_TERMINAL_CONFIGURE ON
  USES_TERMINAL_BUILD ON
  USES_TERMINAL_INSTALL ON
  BUILD_ALWAYS ${BUILD_ALWAYS_TOOLS}
)

ExternalProject_Add_Step(SDL_shadercross
  DownloadDirectXShaderCompiler
  COMMAND ${CMAKE_COMMAND} -P <SOURCE_DIR>/build-scripts/download-prebuilt-DirectXShaderCompiler.cmake -DDXC_ROOT=${CMAKE_CURRENT_BINARY_DIR}
  DEPENDERS configure
  USES_TERMINAL ON
  LOG ON
  ALWAYS ON
)

ExternalProject_Add(
  protobuf
  SOURCE_DIR ${THIRD_PARTY_SRC_DIR}/protobuf-3.19.1
  CMAKE_ARGS
    -Dprotobuf_BUILD_TESTS:BOOL=OFF
    -Dprotobuf_BUILD_EXAMPLES:BOOL=OFF
    -DBUILD_SHARED_LIBS:BOOL=OFF
    ${COMMON_FLAGS}
  SOURCE_SUBDIR cmake
  USES_TERMINAL_CONFIGURE ON
  USES_TERMINAL_BUILD ON
  USES_TERMINAL_INSTALL ON
  BUILD_ALWAYS ${BUILD_ALWAYS_TOOLS}
)

create_import_cmake_file(PROJECTS
    SDL3
    SPIRV-Cross
    SDL_shadercross
    protobuf
)
